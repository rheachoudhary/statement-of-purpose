% Shared SOP preamble, used by Rhea_SOP.tex and every sop/<school>/ file.
% Each school file defines \school and \program before \input-ing this;
% \schoolheader (header line) defaults to \school.
\documentclass[11pt]{article}
\usepackage{lmodern}          % AER font
\usepackage[T1]{fontenc}      % Better font encoding
\usepackage{amsmath, amssymb}
\usepackage[margin=1in]{geometry}
\usepackage{graphicx}
\usepackage{url}
\usepackage[round, longnamesfirst]{natbib}
\usepackage{hyperref}

%=======================================
% Format helpers
%=======================================
% Target school -- used in the header and the first paragraph.
\providecommand{\school}{[University name]}
\providecommand{\program}{[Program name]}
\providecommand{\schoolheader}{\school}

% Run-in section heading: bold label ending in a period, body text follows
% on the same line (matches the reference format).
\newcommand{\shead}[1]{\par\vspace{0.85em}\noindent\textbf{#1.}\ }

% Indented paragraphs, no extra vertical space between them.
\setlength{\parindent}{1.5em}
\setlength{\parskip}{0pt}

% Word count footer. The VS Code build runs texcount first, writing the count
% to \jobname.wc; it shows "??" until texcount has run once.
\usepackage{fancyhdr}
\newread\wcfile
\IfFileExists{\jobname.wc}{\openin\wcfile=\jobname.wc \read\wcfile to\wcnum \closein\wcfile}{\def\wcnum{??}}
\pagestyle{fancy}\fancyhf{}\renewcommand{\headrulewidth}{0pt}
\fancyfoot[C]{\thepage}\fancyfoot[R]{\small Word count: \wcnum}

% Compact reference list: small font, tight spacing, heading styled like \shead.
\renewcommand{\bibsection}{\par\vspace{0.85em}\noindent\textbf{References}\par\vspace{0.3em}}
\renewcommand{\bibfont}{\small}
\setlength{\bibsep}{1pt}
\bibliographystyle{abbrvnat}
